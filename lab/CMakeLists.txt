cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(DriftVMLab VERSION 0.2.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  include(CTest)
endif()
set(LAB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
execute_process(COMMAND git rev-parse --short=12 HEAD WORKING_DIRECTORY "${LAB_ROOT}"
  OUTPUT_VARIABLE LAB_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT LAB_REVISION)
  set(LAB_REVISION "unversioned")
endif()
add_executable(driftvm_lab "${LAB_ROOT}/src/lab.cpp")
target_compile_definitions(driftvm_lab PRIVATE DRIFTVM_REVISION="${LAB_REVISION}")
if(BUILD_TESTING)
  add_executable(driftvm_module_tests "${LAB_ROOT}/tests/modules_tests.cpp")
  add_test(NAME modules-core COMMAND driftvm_module_tests "${LAB_ROOT}/examples/min-transition.json")
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_FOUND)
    add_test(NAME modules-integration COMMAND "${Python3_EXECUTABLE}" "${LAB_ROOT}/tests/lab_integration.py" $<TARGET_FILE:driftvm_lab> $<TARGET_FILE:driftvm_module_tests>)
  endif()
endif()
foreach(target driftvm_lab driftvm_module_tests)
  if(TARGET ${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
  endif()
endforeach()
