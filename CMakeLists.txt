cmake_minimum_required(VERSION 3.20)
project(DriftVM VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
execute_process(COMMAND git rev-parse --short=12 HEAD WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
  OUTPUT_VARIABLE DRIFTVM_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT DRIFTVM_REVISION)
  set(DRIFTVM_REVISION "unversioned")
endif()
add_executable(driftvm src/main.cpp)
target_compile_definitions(driftvm PRIVATE DRIFTVM_REVISION="${DRIFTVM_REVISION}")
include(CTest)
if(BUILD_TESTING)
  add_executable(driftvm_tests tests/core_tests.cpp)
  add_test(NAME core-regressions COMMAND driftvm_tests)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_FOUND)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:driftvm>)
  endif()
endif()
foreach(target driftvm driftvm_tests)
  if(TARGET ${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
  endif()
endforeach()
add_subdirectory(lab)
