cmake_minimum_required(VERSION 3.20)
project(DriftSort VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(driftsort search.cpp)
if(MSVC)
  target_compile_options(driftsort PRIVATE /W4 /permissive-)
else()
  target_compile_options(driftsort PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
add_test(NAME sort-core COMMAND driftsort --self-test)
find_package(Python3 COMPONENTS Interpreter)
if(BUILD_TESTING AND Python3_FOUND)
  add_test(NAME sort-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests.py $<TARGET_FILE:driftsort>)
endif()
